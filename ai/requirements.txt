# 1) 가상환경 생성 및 활성화 (권장)
#    python -m venv .venv
#    .\.venv\Scripts\Activate.ps1        # Windows PowerShell
#    # source .venv/bin/activate         # macOS / Linux
#
# 2) 의존성 설치
#    pip install -r requirements.txt
#
# 3) 원천 데이터 준비
#    data/raw/ 에 사고·인구 CSV를 배치하세요.
#    (예: 사고분석.csv — Git에 포함되지 않음)
#
# 4) 전처리 및 모델 학습 (최초 1회, .pkl 생성용)
#    python -m src.preprocess
#    python -m src.train
#
# 5) API 서버 실행
#    uvicorn app.main:app --reload --host 0.0.0.0 --port 8000
#
#    서버 주소 예: http://localhost:8000
#    API 문서: http://localhost:8000/docs
#
# --- 기타 자주 쓰는 명령어 ---
#    python -m src.inference --구군 달서구 --연령대 "51-60세" --성별 남 --차종 승용
#    docker build -t ai-traffic-risk .
#    docker run -p 8000:8000 ai-traffic-risk
#
# 참고:
# - 이 프로젝트는 Python 기반입니다. npm install이 아니라 pip install을 사용합니다.
